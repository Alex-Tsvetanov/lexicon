# English lexicon of the film domain.
#
# PARADIGM <name>            opens a paradigm
# FORM <affix> [feat=value]  one form of that paradigm
#     affix syntax:  +          the bare stem
#                    +s         append "s"
#                    ~1+ies     drop one character, then append "ies"
# LEX <stem> <lemma> <pos> <paradigm> [sem field]
#     sem fields:  type=<T>    the noun denotes entities of type T
#                  rel=<R>     the verb denotes relation R
#                  objprep=<p> the object of the verb arrives through preposition p
#                  ent=<id>    the proper noun denotes this entity
#                  prep=<p>    preposition, resolved through the frames in schema.txt
#                  det=<q>     determiner and the quantifier it introduces
#                  wh=<k>      question word
#
# The paradigms and the lexicon are composed into one transducer at load time.

PARADIGM NOUN_S
FORM +          num=sg
FORM +s         num=pl

PARADIGM NOUN_ES
FORM +          num=sg
FORM +es        num=pl

PARADIGM NOUN_Y
FORM +          num=sg
FORM ~1+ies     num=pl

PARADIGM VERB_REG
FORM +          vform=base
FORM +s         vform=fin tense=pres num=sg pers=3
FORM +ed        vform=past tense=past
FORM +ed        vform=part
FORM +ing       vform=ger

PARADIGM VERB_E
FORM +          vform=base
FORM +s         vform=fin tense=pres num=sg pers=3
FORM ~1+ed      vform=past tense=past
FORM ~1+ed      vform=part
FORM ~1+ing     vform=ger

# "person" has an irregular plural, which the affix syntax expresses by deleting
# the whole stem: the alternation is data, not a special case in the code.
PARADIGM NOUN_PEOPLE
FORM +          num=sg
FORM ~6+people  num=pl

PARADIGM INVAR
FORM +          -

# --- common nouns ----------------------------------------------------------
LEX person      person      noun    NOUN_PEOPLE type=Person
LEX film        film        noun    NOUN_S      type=Film
LEX movie       movie       noun    NOUN_S      type=Film
LEX actor       actor       noun    NOUN_S      type=Person
LEX actress     actress     noun    NOUN_ES     type=Person
LEX director    director    noun    NOUN_S      type=Person
LEX studio      studio      noun    NOUN_S      type=Studio
LEX country     country     noun    NOUN_Y      type=Country

# The first verb of a relation is the one the synthesiser uses to say it, so
# "direct" is listed before the verb "film".
LEX direct      direct      verb    VERB_REG    rel=directed
# "films" is also a verb form. The homonymy is real and the morphological layer
# keeps both readings; the parser is what decides between them.
LEX film        film        verb    VERB_REG    rel=directed
LEX produce     produce     verb    VERB_E      rel=produced
LEX act         act         verb    VERB_REG    rel=acted_in objprep=in
